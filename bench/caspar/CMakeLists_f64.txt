cmake_minimum_required(VERSION 3.20)
project(caspar_bal LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB CASPAR_SRC gen/*.cu gen/*.cc)
add_executable(caspar_bal caspar_bal.cc ${CASPAR_SRC})
target_include_directories(caspar_bal PRIVATE gen /usr/include/eigen3 /usr/local/cuda/include)
set_target_properties(caspar_bal PROPERTIES CUDA_ARCHITECTURES "89" CUDA_SEPARABLE_COMPILATION ON)
target_compile_options(caspar_bal PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3> $<$<COMPILE_LANGUAGE:CXX>:-O2>)
target_link_libraries(caspar_bal cudart)
